CREATE: CREACIÓN DE TABLAS NORMALIZADAS (1FN, 2FN, 3FN y 4FN) -------------------------------------------------------------------------------- -- Tabla Clientes (3FN: Sin dependencias transitivas) CREATE TABLE clientes ( id_cliente NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, nombre VARCHAR2(100) NOT NULL, email VARCHAR2(100) UNIQUE NOT NULL ); -- Tabla Productos (1FN y 2FN: Valores atómicos y llave primaria definida) CREATE TABLE productos ( id_producto NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, nombre_producto VARCHAR2(100) NOT NULL, precio NUMBER(10,2) NOT NULL ); -- Tabla Pedidos (Cabecera del pedido) CREATE TABLE pedidos ( id_pedido NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, id_cliente NUMBER NOT NULL, fecha_pedido DATE NOT NULL, CONSTRAINT fk_pedido_cliente FOREIGN KEY (id_cliente) REFERENCES clientes(id_cliente) ); -- Tabla Detalle de Pedidos (Resuelve relaciones N:M - 2FN) CREATE TABLE detalle_pedidos ( id_pedido NUMBER NOT NULL, id_producto NUMBER NOT NULL, cantidad NUMBER NOT NULL, CONSTRAINT pk_detalle_pedidos PRIMARY KEY (id_pedido, id_producto), CONSTRAINT fk_detalle_pedido FOREIGN KEY (id_pedido) REFERENCES pedidos(id_pedido) ON DELETE CASCADE, CONSTRAINT fk_detalle_producto FOREIGN KEY (id_producto) REFERENCES productos(id_producto) ); -- Tabla Asignación Proveedor-Producto (4FN: Descomposición de dependencias multivaluadas) CREATE TABLE asignacion_proveedor_producto ( id_producto NUMBER NOT NULL, proveedor VARCHAR2(100) NOT NULL, CONSTRAINT pk_asig_prov_prod PRIMARY KEY (id_producto, proveedor), CONSTRAINT fk_app_producto FOREIGN KEY (id_producto) REFERENCES productos(id_producto) ON DELETE CASCADE ); -------------------------------------------------------------------------------- -- 2. SELECT: CONSULTAS Y VISTAS (Unión relacional de las tablas normalizadas) -------------------------------------------------------------------------------- -- Consulta directa uniendo las tablas SELECT p.id_pedido, c.nombre AS cliente, c.email, p.fecha_pedido, pr.nombre_producto, dp.cantidad, pr.precio, (dp.cantidad * pr.precio) AS subtotal FROM pedidos p JOIN clientes c ON p.id_cliente = c.id_cliente JOIN detalle_pedidos dp ON p.id_pedido = dp.id_pedido JOIN productos pr ON dp.id_producto = pr.id_producto; -- Creación de la Vista oficial para respaldar reportes CREATE OR REPLACE VIEW vista_detalle_pedidos AS SELECT p.id_pedido, c.nombre AS cliente, c.email, p.fecha_pedido, pr.nombre_producto, dp.cantidad, pr.precio FROM pedidos p JOIN clientes c ON p.id_cliente = c.id_cliente JOIN detalle_pedidos dp ON p.id_pedido = dp.id_pedido JOIN productos pr ON dp.id_producto = pr.id_producto; -- Probar la vista con un SELECT SELECT * FROM vista_detalle_pedidos; -------------------------------------------------------------------------------- -- 4.
